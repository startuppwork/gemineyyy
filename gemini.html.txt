    <script>
        // --- Global Variables & Dummy Data ---
        let citizenMap, hotspotMap;
        let safeRoutePolyline;
        let currentUser = null;
        let userTokens = 0;
        const DUMMY_AQI_DATA = [
            { lat: 26.9124, lng: 75.7873, aqi: 155, name: "City Palace" },
            { lat: 26.853, lng: 75.805, aqi: 210, name: "Jawahar Circle" },
            { lat: 26.924, lng: 75.827, aqi: 85, name: "Hawa Mahal" },
            { lat: 26.935, lng: 75.755, aqi: 320, name: "Vaishali Nagar" },
            { lat: 26.828, lng: 75.806, aqi: 120, name: "Malviya Nagar" },
        ];
        
        // --- Core Application Logic ---
        document.addEventListener('DOMContentLoaded', function() {
            const loginForm = document.getElementById('login-form');
            const logoutBtn = document.getElementById('logout-btn');

            loginForm.addEventListener('submit', handleLogin);
            logoutBtn.addEventListener('click', handleLogout);

            // Check for logged-in user in session storage (for page refresh)
            const savedUser = sessionStorage.getItem('airNexusUser');
            if (savedUser) {
                currentUser = JSON.parse(savedUser);
                showDashboard(currentUser.role);
            }
        });

        // --- Authentication ---
        function handleLogin(e) {
            e.preventDefault();
            const username = document.getElementById('username').value.trim();
            const password = document.getElementById('password').value;
            const loginAlert = document.getElementById('login-alert');

            if (username === 'citizen' && password === 'password') {
                currentUser = { name: 'Citizen User', role: 'citizen' };
                sessionStorage.setItem('airNexusUser', JSON.stringify(currentUser));
                showDashboard('citizen');
            } else if (username === 'policymaker' && password === 'password') {
                currentUser = { name: 'Policy Maker', role: 'policymaker' };
                sessionStorage.setItem('airNexusUser', JSON.stringify(currentUser));
                showDashboard('policymaker');
            } else {
                loginAlert.textContent = 'Invalid credentials. Please try again.';
                loginAlert.classList.remove('d-none');
            }
        }

        function handleLogout() {
            currentUser = null;
            sessionStorage.removeItem('airNexusUser');
            switchView('login-view');
            document.getElementById('logout-btn').classList.add('d-none');
            document.getElementById('user-greeting').classList.add('d-none');
            // Clean up maps and charts if necessary
            if (citizenMap) citizenMap.remove();
            if (hotspotMap) hotspotMap.remove();
        }

        function showDashboard(role) {
            document.getElementById('logout-btn').classList.remove('d-none');
            const userGreeting = document.getElementById('user-greeting');
            userGreeting.textContent = `Welcome, ${currentUser.name}`;
            userGreeting.classList.remove('d-none');
            
            if (role === 'citizen') {
                switchView('citizen-view');
                initCitizenDashboard();
            } else if (role === 'policymaker') {
                switchView('policymaker-view');
                initPolicymakerDashboard();
            }
        }

        function switchView(viewId) {
            document.querySelectorAll('.view').forEach(view => view.classList.remove('active-view'));
            document.getElementById(viewId).classList.add('active-view');
        }

        // --- Citizen Mode Functions ---
        function initCitizenDashboard() {
            // Use a timeout to ensure the map container is visible before initialization
            setTimeout(() => {
                initializeCitizenMap();
                populateAqiMarkers();
            }, 100);
            
            document.getElementById('vehicle-form').addEventListener('submit', handleVehicleProfile);
            document.getElementById('safe-route-btn').addEventListener('click', showSafeRoute);
        }

        function initializeCitizenMap() {
            if (citizenMap) return; // Already initialized
            citizenMap = L.map('map').setView([26.9124, 75.7873], 12); // Centered on Jaipur
            L.tileLayer('https://{s}.tile.openstreetmap.org/{z}/{x}/{y}.png', {
                attribution: '&copy; <a href="https://www.openstreetmap.org/copyright">OpenStreetMap</a> contributors'
            }).addTo(citizenMap);
        }

        function getAqiColor(aqi) {
            if (aqi <= 50) return 'green';
            if (aqi <= 100) return 'yellow';
            if (aqi <= 200) return 'orange';
            if (aqi <= 300) return 'red';
            return 'purple';
        }

        function populateAqiMarkers() {
            let highestAqi = 0;
            DUMMY_AQI_DATA.forEach(point => {
                if(point.aqi > highestAqi) highestAqi = point.aqi;

                const icon = L.divIcon({
                    className: 'custom-div-icon',
                    html: `<div style='background-color:${getAqiColor(point.aqi)};' class='marker-pin'></div><i class='bi bi-wind'></i>`,
                    iconSize: [30, 42],
                    iconAnchor: [15, 42]
                });
                L.marker([point.lat, point.lng])
                 .addTo(citizenMap)
                 .bindPopup(`<b>${point.name}</b><br>AQI: ${point.aqi}`);
            });
            
            // Health Alert
            const healthAlert = document.getElementById('health-alert');
            if(highestAqi > 150) {
                healthAlert.innerHTML = `<i class="bi bi-exclamation-triangle-fill"></i> <strong>Health Alert:</strong> Unhealthy AQI levels (${highestAqi}) detected nearby. Limit outdoor activity.`;
                healthAlert.classList.remove('d-none');
            } else {
                 healthAlert.classList.add('d-none');
            }
        }

        function showSafeRoute() {
            if (safeRoutePolyline) {
                citizenMap.removeLayer(safeRoutePolyline);
            }
            // Dummy route that avoids high AQI points (Vaishali Nagar & Jawahar Circle)
            const latlngs = [
                [26.924, 75.827], // Hawa Mahal (Good)
                [26.9124, 75.7873], // City Palace (Moderate)
                [26.828, 75.806]  // Malviya Nagar (Moderate)
            ];
            safeRoutePolyline = L.polyline(latlngs, {color: 'blue', dashArray: '5, 10'}).addTo(citizenMap);
            citizenMap.fitBounds(safeRoutePolyline.getBounds());
            alert('A suggested safer route is now shown on the map in blue.');
        }

        function handleVehicleProfile(e) {
            e.preventDefault();
            const profile = {
                name: document.getElementById('vehicleName').value,
                type: document.getElementById('vehicleType').value,
                fuel: document.getElementById('fuelType').value,
                age: document.getElementById('vehicleAge').value,
                std: document.getElementById('emissionStd').value,
                km: document.getElementById('weeklyKm').value,
            };
            
            const display = document.getElementById('vehicle-profile-display');
            const icon = profile.type === 'Car' ? 'bi-car-front-fill' : 'bi-bicycle';
            
            display.innerHTML = `
                <div class="vehicle-card">
                    <i class="bi ${icon}" style="font-size: 4rem; color: #0d6efd;"></i>
                    <h5>${profile.name}</h5>
                    <p>${profile.age} years old | ${profile.km} km/week</p>
                    <div>
                        <span class="badge bg-primary">${profile.type}</span>
                        <span class="badge bg-info text-dark">${profile.fuel}</span>
                        <span class="badge bg-secondary">${profile.std}</span>
                    </div>
                </div>
            `;
            
            // Close modal
            const modal = bootstrap.Modal.getInstance(document.getElementById('vehicleModal'));
            modal.hide();
        }

        function earnTokens(amount) {
            userTokens += amount;
            document.getElementById('token-count').textContent = userTokens;
            alert(`Congratulations! You've earned ${amount} tokens.`);
        }

        // --- Policymaker Mode Functions ---
        function initPolicymakerDashboard() {
             setTimeout(() => {
                renderSourceChart();
                renderTrendsChart();
                initializeHotspotMap();
            }, 100);
        }

        function renderSourceChart() {
            const ctx = document.getElementById('source-chart').getContext('2d');
            new Chart(ctx, {
                type: 'doughnut',
                data: {
                    labels: ['Vehicular Emissions', 'Industrial Pollution', 'Construction Dust', 'Stubble Burning', 'Other'],
                    datasets: [{
                        label: 'Pollution Source Contribution',
                        data: [40, 25, 15, 12, 8],
                        backgroundColor: ['#dc3545', '#6c757d', '#ffc107', '#fd7e14', '#198754'],
                        borderColor: '#fff',
                        borderWidth: 2
                    }]
                },
                options: {
                    responsive: true,
                    maintainAspectRatio: false,
                    plugins: {
                        legend: { position: 'top' },
                        title: { display: true, text: 'Pollution Source Contribution (%)' }
                    }
                }
            });
        }
        
        function renderTrendsChart() {
             const ctx = document.getElementById('trends-chart').getContext('2d');
             new Chart(ctx, {
                type: 'line',
                data: {
                    labels: ['-72h', '-48h', '-24h', 'Now', '+24h', '+48h', '+72h'],
                    datasets: [{
                        label: 'AQI PM2.5',
                        data: [120, 135, 180, 155, 190, 210, 170],
                        fill: true,
                        borderColor: '#0d6efd',
                        backgroundColor: 'rgba(13, 110, 253, 0.1)',
                        tension: 0.4
                    }, {
                        label: 'Forecast Confidence',
                        data: [null, null, null, 155, 160, 175, 140],
                        borderColor: '#adb5bd',
                        borderDash: [5, 5],
                        label: 'Lower Bound'
                    },
                    {
                        label: 'Forecast Confidence',
                         data: [null, null, null, 155, 220, 245, 200],
                        fill: '-1',
                        backgroundColor: 'rgba(173, 181, 189, 0.2)',
                        borderColor: '#adb5bd',
                        borderDash: [5, 5],
                         label: 'Upper Bound'
                    }]
                },
                options: {
                    responsive: true,
                    maintainAspectRatio: false,
                    plugins: {
                        legend: { display: false },
                        title: { display: true, text: 'Historical and Forecasted AQI' }
                    },
                    scales: { y: { beginAtZero: true } }
                }
             });
        }
        
        function initializeHotspotMap() {
            if (hotspotMap) return;
            hotspotMap = L.map('hotspot-map').setView([26.9124, 75.7873], 11);
            L.tileLayer('https://{s}.basemaps.cartocdn.com/dark_all/{z}/{x}/{y}{r}.png', {
                attribution: '&copy; <a href="https://www.openstreetmap.org/copyright">OpenStreetMap</a> contributors &copy; <a href="https://carto.com/attributions">CARTO</a>',
            }).addTo(hotspotMap);

            // Add dummy heatmap data (circles for simplicity)
            DUMMY_AQI_DATA.forEach(point => {
                if (point.aqi > 150) { // Only show hotspots
                    L.circle([point.lat, point.lng], {
                        color: getAqiColor(point.aqi),
                        fillColor: getAqiColor(point.aqi),
                        fillOpacity: 0.4,
                        radius: point.aqi * 10 // Radius proportional to AQI
                    }).addTo(hotspotMap).bindPopup(`<b>${point.name}</b><br>AQI: ${point.aqi}<br>Est. Population Affected: ${point.aqi * 150}`);
                }
            });
        }

    </script>
